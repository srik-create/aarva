# Aarva v0.1 — runtime dependencies.
#
# Install on a fresh venv with:
#   python3.12 -m venv .venv
#   source .venv/bin/activate
#   pip install --upgrade pip
#   pip install -r requirements.txt
#
# Tested against Python 3.12 on macOS. We pin 3.12 in pyproject.toml +
# the launchd wrapper for predictability. The previous Kokoro/onnxruntime
# constraint that required 3.10+ was removed when Gemini TTS replaced
# Kokoro; 3.10+ is still a safe floor.
#
# Pinning policy: lower bound to the version we know works, no upper
# bound (so security patches and minor versions just flow). Revisit if
# anything breaks after a major version bump.

# ─── Ingestion ────────────────────────────────────────────────────────────
feedparser>=6.0            # RSS/Atom feed parsing (Stage 1)
httpx>=0.27                # HTTP client (RSS fetching + article extraction)
trafilatura>=2.0           # Article-body extraction; v2 API for favor_recall / no_fallback
python-dateutil>=2.8       # Lenient date parsing in RSS entries
trendspyg>=1.5             # Google Trends RSS path (trend signal, Stage 3 crawl —
                           # pytrends is archived/unmaintained, verified 2026-08-13)

# ─── Stage 1.5 consolidation ──────────────────────────────────────────────
numpy>=1.26                # Embedding vector math
scikit-learn>=1.4          # TF-IDF fallback path; cosine_similarity
# sentence-transformers — removed from the runtime image 2026-06-30.
# The production embedding backend moved to Vertex AI (see
# pipeline.yaml's `embedding:` block). The LocalEmbeddingClient in
# aarva/clients/embedding.py is still in-tree as an offline-dev
# fallback; install sentence-transformers manually if you flip
# `embedding.provider` back to `local`:
#     pip install 'sentence-transformers>=3.0'
# It was removed because PyTorch + the BGE-base model push past
# Render Starter's 512 MB RAM ceiling when loaded on first prompt.

# ─── LLM clients (Stage 4+5+6, Stage 8a/b, Stage 9 narrator detection) ────
# google-genai is the production default per project policy
# (see aarva/clients/llm.py and pipeline.yaml).
# anthropic is kept for debug/fallback only.
google-genai>=1.0          # Gemini API (production LLM)
anthropic>=0.40            # Anthropic API (fallback)

# ─── Stage 9 TTS ──────────────────────────────────────────────────────────
# Gemini TTS is the production path. It uses the same `google-genai`
# package already pinned above for LLM calls — no extra dep needed.
# Audio handling uses Python's stdlib `wave` module. Removed when
# Gemini TTS landed: kokoro-onnx, onnxruntime.

# ─── Config / utilities ───────────────────────────────────────────────────
pyyaml>=6.0                # All config files
click>=8.1                 # daily.py CLI

# ─── Asset generation ────────────────────────────────────────────────────
Pillow>=10.0               # scripts/generate_logo.py (podcast cover)

# ─── Doc reading (audit + diagnostic scripts) ────────────────────────────
python-docx>=1.0           # Reads docs/aarva_kickoff.docx in scripts that
                           # cross-reference the kickoff against code state

# ─── Audio CDN (Stage 10 publish) ─────────────────────────────────────────
# Cloudflare R2 is S3-compatible, so boto3 (AWS SDK) talks to it cleanly.
# Used by aarva/output/r2_uploader.py to upload MP3s after conversion so
# RSS <enclosure> URLs point at R2 rather than GitHub Pages (which has a
# 1 GB site-size soft cap that Aarva would hit in ~15 days at current
# volume). R2 has 10 GB free + zero egress fees, so this is permanently
# cheap for podcast hosting.
boto3>=1.35

# ─── Web app (aarva/server/) ──────────────────────────────────────────────
# Standard ASGI Python stack. Provider-agnostic: the same image runs on
# Render today, Fly.io / Railway / DO / VPS tomorrow (per AGENTS.md
# rule 7b — design for portability by default).
fastapi>=0.110
uvicorn[standard]>=0.27        # ASGI server; [standard] adds httptools + uvloop
jinja2>=3.1                    # HTML templates
python-multipart>=0.0.9        # Form parsing (email collection on episode-create)

# ─── Email (listener "your episode is ready" notifications) ──────────────
# Used by aarva/services/email.py. The service falls back to a stub
# (logs the would-send body) when RESEND_API_KEY is unset, so the SDK
# being installed doesn't force email-on. Domain verification + DKIM
# happen in the Resend dashboard against aarva.app.
resend>=2.0
